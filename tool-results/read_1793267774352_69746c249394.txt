   979→/** Remplace les jetons {nom} par les valeurs fournies (sans toucher aux inconnus). */
   980→function interpolate(template: string, vars?: I18nVars): string {
   981→  if (!vars) return template;
   982→  return template.replace(/\{(\w+)\}/g, (match, name: string) =>
   983→    Object.prototype.hasOwnProperty.call(vars, name) ? String(vars[name]) : match
   984→  );
   985→}
   986→
   987→/**
   988→ * Accès i18n de la boutique : `lang` courant (fr pendant SSR + hydratation, puis
   989→ * langue persistée), `setLang` (persisté par zustand) et `t(clé, vars?)` avec
   990→ * repli en cascade : dict[lang] → dict.fr → la clé elle-même.
   991→ */
   992→export function useI18n(): {
   993→  lang: Lang;
   994→  setLang: (l: Lang) => void;
   995→  t: (key: string, vars?: I18nVars) => string;
   996→  mounted: boolean;
   997→} {
   998→  const { lang, setLang, mounted } = useLang();
   999→  const t = useCallback(
  1000→    (key: string, vars?: I18nVars): string => {
  1001→      const entry = DICTS[lang][key] ?? DICTS.fr[key];
  1002→      if (entry === undefined) return key;
  1003→      return typeof entry === "function" ? entry(vars ?? {}) : interpolate(entry, vars);
  1004→    },
  1005→    [lang]
  1006→  );
  1007→  return { lang, setLang, t, mounted };
  1008→}
