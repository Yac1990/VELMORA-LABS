    88→function slugify(s: string): string {
    89→  return s.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "")
    90→    .replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "").slice(0, 80);
    91→}
    92→
    93→function eurosToCents(raw: string): number | null {
    94→  const t = raw.trim().replace(",", ".");
    95→  if (!t) return null;
    96→  const v = Number.parseFloat(t);
    97→  if (Number.isNaN(v) || v < 0) return null;
    98→  return Math.round(v * 100);
    99→}
   100→
   101→function parseJsonArrayField(raw: string | null | undefined): string[] {
   102→  if (!raw) return [];
   103→  try { const v = JSON.parse(raw); return Array.isArray(v) ? v.map(String) : []; } catch { return []; }
   104→}

   106→function StatusBadge({ status }: { status: string }) {
   107→  const s = PRODUCT_STATUSES[status] ?? { label: status, cls: "border-zinc-200 bg-zinc-100 text-zinc-600" };
   108→  return <Badge variant="outline" className={`shrink-0 ${s.cls}`}>{s.label}</Badge>;
   109→}
   110→
   111→function UrlCheckBadge({ status }: { status: string | null }) {
   112→  if (status === "ok") return <Badge variant="outline" className="shrink-0 border-emerald-200 bg-emerald-50 text-emerald-700">URL vérifiée</Badge>;
   113→  if (status === "broken") return <Badge variant="outline" className="shrink-0 border-red-200 bg-red-50 text-red-700">Lien cassé</Badge>;
   114→  return <Badge variant="outline" className="shrink-0 border-zinc-200 bg-zinc-100 text-zinc-600">Non vérifié</Badge>;
   115→}

   802→                    <Select value={initialOffer.partnerId || "none"} onValueChange={(v) => setInitialOffer((s) => ({ ...s, partnerId: v === "none" ? "" : v }))}>
   803→                      <SelectTrigger><SelectValue placeholder="Partenaire" /></SelectTrigger>
   804→                      <SelectContent>
   805→                        <SelectItem value="none">— Aucune —</SelectItem>
   806→                        {partners.map((p) => <SelectItem key={p.id} value={p.id}>{p.name}{p.isDemo ? " (démo)" : ""}</SelectItem>)}
   807→                      </SelectContent>
   808→                    </Select>
   809→                    <Input className="sm:col-span-2" placeholder="URL de destination (http…)" value={initialOffer.destinationUrl}
   810→                      onChange={(e) => setInitialOffer((s) => ({ ...s, destinationUrl: e.target.value }))} />
   811→                    <Input inputMode="decimal" placeholder={`Prix (${form.currency})`} value={initialOffer.price}
   812→                      onChange={(e) => setInitialOffer((s) => ({ ...s, price: e.target.value }))} />
   813→                  </div>
   814→                  <p className="text-xs text-muted-foreground">Une fois le produit créé, gérez toutes ses offres dans la fenêtre d&apos;édition.</p>
   815→                </div>
   816→              )}
   817→            </div>
   818→          )}
   819→          <DialogFooter>
   820→            <Button variant="outline" onClick={() => setDialogOpen(false)} disabled={saving}>Annuler</Button>
   821→            <Button onClick={submitProduct} disabled={saving || detailLoading}>
   822→              {saving && <Loader2 className="h-4 w-4 animate-spin" />}
   823→              {editingId ? "Enregistrer" : "Créer le produit"}
   824→            </Button>
   825→          </DialogFooter>
   826→        </DialogContent>
   827→      </Dialog>
   828→
   829→      {/* Confirmation suppression produit */}
   830→      <AlertDialog open={!!deleteTarget} onOpenChange={(o) => { if (!o) setDeleteTarget(null); }}>
   831→        <AlertDialogContent>
   832→          <AlertDialogHeader>
   833→            <AlertDialogTitle>Supprimer ce produit ?</AlertDialogTitle>
   834→            <AlertDialogDescription>
   835→              « {deleteTarget?.title} » sera définitivement supprimé. S&apos;il a un historique de clics, il sera archivé au lieu d&apos;être supprimé.
   836→            </AlertDialogDescription>
   837→          </AlertDialogHeader>
   838→          <AlertDialogFooter>
   839→            <AlertDialogCancel disabled={deleting}>Annuler</AlertDialogCancel>
   840→            <AlertDialogAction onClick={(e) => { e.preventDefault(); confirmDelete(); }} className="bg-red-600 hover:bg-red-700">
   841→              {deleting ? <Loader2 className="h-4 w-4 animate-spin" /> : <TriangleAlert className="h-4 w-4" />} Supprimer
   842→            </AlertDialogAction>
   843→          </AlertDialogFooter>
   844→        </AlertDialogContent>
   845→      </AlertDialog>
   846→
   847→      {/* Confirmation suppression offre */}
   848→      <AlertDialog open={!!offerDeleteTarget} onOpenChange={(o) => { if (!o) setOfferDeleteTarget(null); }}>
   849→        <AlertDialogContent>
   850→          <AlertDialogHeader>
   851→            <AlertDialogTitle>Supprimer cette offre ?</AlertDialogTitle>
   852→            <AlertDialogDescription>
   853→              Redirection partenaire « {offerDeleteTarget?.partnerName} ». Si des clics y sont rattachés, elle sera simplement désactivée.
   854→            </AlertDialogDescription>
   855→          </AlertDialogHeader>
   856→          <AlertDialogFooter>
   857→            <AlertDialogCancel disabled={offersBusy}>Annuler</AlertDialogCancel>
   858→            <AlertDialogAction onClick={(e) => { e.preventDefault(); confirmDeleteOffer(); }} className="bg-red-600 hover:bg-red-700">
   859→              {offersBusy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Trash2 className="h-4 w-4" />} Supprimer
   860→            </AlertDialogAction>
   861→          </AlertDialogFooter>
   862→        </AlertDialogContent>
   863→      </AlertDialog>
   864→
   865→      {!canWrite && (
   866→        <Alert>
   867→          <Package className="h-4 w-4" />
   868→          <AlertTitle>Lecture seule</AlertTitle>
   869→          <AlertDescription>Votre rôle ne permet pas de modifier le catalogue — les boutons d&apos;écriture sont masqués et le serveur refuserait toute modification.</AlertDescription>
   870→        </Alert>
   871→      )}
   872→    </div>
   873→  );
